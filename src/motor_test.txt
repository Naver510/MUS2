#include <Arduino.h>

// DRV8835 pin map matches main.cpp so the driver wiring stays consistent.
#define MODE_PIN 21
#define AIN1_PIN 18
#define AIN2_PIN 17
#define BIN1_PIN 16
#define BIN2_PIN 15

// Set true if motor B spins the opposite way relative to motor A when driven identically.
constexpr bool MOTOR_B_INVERTED = true;

static inline void driveMotorA(int speed) {
	if (speed >= 0) {
		digitalWrite(AIN1_PIN, HIGH);
		analogWrite(AIN2_PIN, speed);
	} else {
		digitalWrite(AIN1_PIN, LOW);
		analogWrite(AIN2_PIN, -speed);
	}
}

static inline void driveMotorB(int speed) {
	// Invert phase if wiring makes "forward" spin backwards.
	const bool forwardPhaseHigh = !MOTOR_B_INVERTED;
	int magnitude = speed >= 0 ? speed : -speed;
	bool phase = (speed >= 0) ? forwardPhaseHigh : !forwardPhaseHigh;
	analogWrite(BIN2_PIN, magnitude);
	digitalWrite(BIN1_PIN, phase ? HIGH : LOW);
}

void speedcontrol(int mota, int motb) {
	driveMotorA(mota);
	driveMotorB(motb);
}

void forward_brake(int posa, int posb) {
	driveMotorA(posa);
	driveMotorB(posb);
}

void left_brake(int posa, int posb) {
	// A forward, B reverse
	driveMotorA(posa);
	driveMotorB(-posb);
}

void right_brake(int posa, int posb) {
	// A reverse, B forward
	driveMotorA(-posa);
	driveMotorB(posb);
}

// Run both motors forward for a short window at a low duty cycle.
// Call this from setup() when you want a one-shot direction check.
void runDrv8835ForwardTest(uint8_t speed = 80, uint32_t durationMs = 5000) {
	pinMode(MODE_PIN, OUTPUT);
	pinMode(AIN1_PIN, OUTPUT);
	pinMode(AIN2_PIN, OUTPUT);
	pinMode(BIN1_PIN, OUTPUT);
	pinMode(BIN2_PIN, OUTPUT);

	digitalWrite(MODE_PIN, HIGH); // phase/enable pin high: fast-decay mode per DRV8835 datasheet

	// Forward: AIN1/BIN1 high selects forward phase; AIN2/BIN2 PWM sets speed.
	digitalWrite(AIN1_PIN, HIGH);
	digitalWrite(BIN1_PIN, HIGH);
	analogWrite(AIN2_PIN, speed);
	analogWrite(BIN2_PIN, speed);

	delay(durationMs);

	// Stop motors after the window.
	analogWrite(AIN2_PIN, 0);
	analogWrite(BIN2_PIN, 0);
	digitalWrite(AIN1_PIN, LOW);
	digitalWrite(BIN1_PIN, LOW);
}

// Minimal Arduino entry points: run the forward test once, then stay idle.
void setup() {
	runDrv8835ForwardTest();
}

void loop() {
	// Nothing to repeat; motors already stopped.
}
